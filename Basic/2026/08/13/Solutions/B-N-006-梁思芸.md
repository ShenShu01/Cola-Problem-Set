#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main()
{
	int N;
	cin>>N;
	vector<int> a(N);
	for(int i = 0;i < N;i++)
	{
		cin>>a[i]; 
	}
	sort(a.begin(),a.end());
	vector<int> res;
	for(int i = 0;i < N;i++)
	{
	if(i==0||a[i]==a[i+1])
	{
	res.push_back(a[i]);
    }
    }
	int cnt = res.size();
	cout << cnt << endl;
	for(int i = 0;i < res.size();i++)
	{
	cout << res[i] << " ";
    }
    return 0;
}
